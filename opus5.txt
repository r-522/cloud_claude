/**
 * モデル定義と Messages API リクエストボディの組み立て。
 *
 * 参照した一次情報（Anthropic 公式ドキュメント）:
 *  - Models overview            : docs.anthropic.com/en/docs/about-claude/models/overview
 *  - Messages API リファレンス   : docs.anthropic.com/en/api/messages
 *  - Extended thinking          : docs.anthropic.com/en/docs/build-with-claude/extended-thinking
 *
 * 注意:
 *  「Sonnet 5 / Opus 5」という名称のモデルは公式の Models overview に存在しないため、
 *  実在する最新世代の ID（Claude 4.5 系）へ置き換えている。
 *  新モデルが公開された場合は、下記 MODELS の `id` を公式 Models overview の
 *  「Model名 / Model ID」表に合わせて更新するだけでよい（他のコード変更は不要）。
 */

export interface ModelDef {
  /** Messages API の `model` に渡す ID */
  id: string;
  /** UI 表示名 */
  label: string;
  /** UI 補足説明 */
  description: string;
  /** モデルの最大出力トークン（安全側の上限として利用） */
  maxOutputTokens: number;
  /** Extended thinking に対応しているか */
  supportsThinking: boolean;
}

export const MODELS: ModelDef[] = [
  {
    id: 'claude-haiku-4-5',
    label: 'Haiku 4.5',
    description: '高速・低コスト。日常的な質問や短い作業向け。',
    maxOutputTokens: 64000,
    supportsThinking: true,
  },
  {
    id: 'claude-sonnet-4-5',
    label: 'Sonnet 4.5',
    description: 'バランス型。長文作成やコーディングの標準選択。',
    maxOutputTokens: 64000,
    supportsThinking: true,
  },
  {
    id: 'claude-opus-4-5',
    label: 'Opus 4.5',
    description: '最も高性能。複雑な推論・設計・分析向け。',
    maxOutputTokens: 64000,
    supportsThinking: true,
  },
];

export interface EffortLevel {
  id: string;
  label: string;
  /** Extended thinking の budget_tokens（0 は thinking 無効） */
  budgetTokens: number;
}

/**
 * UI の「推論量（effort）」。
 * 公式に仕様が確定している `thinking.budget_tokens`（Extended thinking）へマッピングする。
 * 推測に基づく非公開パラメータは送信しない。
 */
export const EFFORT_LEVELS: EffortLevel[] = [
  { id: 'off', label: '標準（思考なし）', budgetTokens: 0 },
  { id: 'low', label: '低', budgetTokens: 2048 },
  { id: 'medium', label: '中', budgetTokens: 6144 },
  { id: 'high', label: '高', budgetTokens: 12288 },
];

/** thinking を使わない場合のベース出力トークン数 */
export const BASE_MAX_OUTPUT_TOKENS = 4096;

export const DEFAULT_SYSTEM_PROMPT =
  'You are a helpful assistant in a chat application. ' +
  'Answer in the same language as the user. ' +
  'Format answers with GitHub-flavored Markdown, and always use fenced code blocks with a language tag for code.';

export function findModel(id: unknown): ModelDef | undefined {
  return MODELS.find((m) => m.id === id);
}

export function findEffort(id: unknown): EffortLevel | undefined {
  return EFFORT_LEVELS.find((e) => e.id === id);
}

export interface ChatMessage {
  role: 'user' | 'assistant';
  content: string;
}

export interface BuildRequestInput {
  model: ModelDef;
  effort: EffortLevel;
  messages: ChatMessage[];
  system?: string;
}

export interface AnthropicRequestBody {
  model: string;
  max_tokens: number;
  stream: true;
  system: string;
  messages: ChatMessage[];
  thinking?: { type: 'enabled'; budget_tokens: number };
  temperature?: number;
}

/**
 * Messages API のリクエストボディを組み立てる。
 * - thinking 有効時は max_tokens > budget_tokens を必ず満たす
 * - thinking 有効時に temperature は指定しない（公式仕様上、非対応のため）
 */
export function buildRequestBody(input: BuildRequestInput): AnthropicRequestBody {
  const { model, effort, messages } = input;
  const useThinking = model.supportsThinking && effort.budgetTokens > 0;

  const maxTokens = useThinking
    ? Math.min(model.maxOutputTokens, BASE_MAX_OUTPUT_TOKENS + effort.budgetTokens)
    : Math.min(model.maxOutputTokens, BASE_MAX_OUTPUT_TOKENS);

  const body: AnthropicRequestBody = {
    model: model.id,
    max_tokens: maxTokens,
    stream: true,
    system: input.system ?? DEFAULT_SYSTEM_PROMPT,
    messages,
  };

  if (useThinking) {
    body.thinking = { type: 'enabled', budget_tokens: effort.budgetTokens };
  } else {
    body.temperature = 1;
  }

  return body;
}

// File: src/auth.ts
/**
 * 簡易アクセス制限用のセッション。
 * HMAC-SHA256 で署名した短命トークンを HttpOnly Cookie に格納する。
 * API キー等の秘密情報はクライアントへ一切渡さない。
 */

const encoder = new TextEncoder();

export const SESSION_COOKIE = 'cc_session';
export const SESSION_TTL_SECONDS = 60 * 60 * 12;

export function timingSafeEqual(a: string, b: string): boolean {
  const ab = encoder.encode(a);
  const bb = encoder.encode(b);
  if (ab.length !== bb.length) return false;
  let diff = 0;
  for (let i = 0; i < ab.length; i++) diff |= ab[i] ^ bb[i];
  return diff === 0;
}

function toBase64Url(bytes: Uint8Array): string {
  let binary = '';
  for (const b of bytes) binary += String.fromCharCode(b);
  return btoa(binary).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
}

async function hmac(secret: string, data: string): Promise<Uint8Array> {
  const key = await crypto.subtle.importKey(
    'raw',
    encoder.encode(secret),
    { name: 'HMAC', hash: 'SHA-256' },
    false,
    ['sign'],
  );
  const sig = await crypto.subtle.sign('HMAC', key, encoder.encode(data));
  return new Uint8Array(sig);
}

export async function createSessionToken(
  secret: string,
  nowMs: number = Date.now(),
  ttlSeconds: number = SESSION_TTL_SECONDS,
): Promise<string> {
  const payload = JSON.stringify({ v: 1, exp: Math.floor(nowMs / 1000) + ttlSeconds });
  const encoded = toBase64Url(encoder.encode(payload));
  const signature = toBase64Url(await hmac(secret, encoded));
  return `${encoded}.${signature}`;
}

export async function verifySessionToken(
  token: string | undefined | null,
  secret: string,
  nowMs: number = Date.now(),
): Promise<boolean> {
  if (!token || !secret) return false;
  const parts = token.split('.');
  if (parts.length !== 2) return false;
  const [encoded, signature] = parts;

  const expected = toBase64Url(await hmac(secret, encoded));
  if (!timingSafeEqual(signature, expected)) return false;

  try {
    const raw = atob(encoded.replace(/-/g, '+').replace(/_/g, '/'));
    const payload = JSON.parse(raw) as { exp?: number };
    if (typeof payload.exp !== 'number') return false;
    return payload.exp * 1000 > nowMs;
  } catch {
    return false;
  }
}

export function parseCookies(header: string | null): Record<string, string> {
  const out: Record<string, string> = {};
  if (!header) return out;
  for (const part of header.split(';')) {
    const idx = part.indexOf('=');
    if (idx < 0) continue;
    const name = part.slice(0, idx).trim();
    const value = part.slice(idx + 1).trim();
    if (name) out[name] = decodeURIComponent(value);
  }
  return out;
}

export function buildSessionCookie(token: string, secure: boolean): string {
  const attrs = [
    `${SESSION_COOKIE}=${token}`,
    'Path=/',
    'HttpOnly',
    'SameSite=Lax',
    `Max-Age=${SESSION_TTL_SECONDS}`,
  ];
  if (secure) attrs.push('Secure');
  return attrs.join('; ');
}

export function clearSessionCookie(secure: boolean): string {
  const attrs = [`${SESSION_COOKIE}=`, 'Path=/', 'HttpOnly', 'SameSite=Lax', 'Max-Age=0'];
  if (secure) attrs.push('Secure');
  return attrs.join('; ');
}

/**
 * Anthropic Messages API 呼び出しと SSE パース。
 * 参照: docs.anthropic.com/en/api/messages, docs.anthropic.com/en/docs/build-with-claude/streaming
 */
import type { AnthropicRequestBody } from './models';

export const ANTHROPIC_API_URL = 'https://api.anthropic.com/v1/messages';
export const ANTHROPIC_VERSION = '2023-06-01';

export function callAnthropic(apiKey: string, body: AnthropicRequestBody): Promise<Response> {
  return fetch(ANTHROPIC_API_URL, {
    method: 'POST',
    headers: {
      'content-type': 'application/json',
      accept: 'text/event-stream',
      'x-api-key': apiKey,
      'anthropic-version': ANTHROPIC_VERSION,
    },
    body: JSON.stringify(body),
  });
}

export interface SSEEvent {
  event: string;
  data: string;
}

/** SSE ストリームを 1 イベントずつ取り出す */
export async function* parseSSE(stream: ReadableStream<Uint8Array>): AsyncGenerator<SSEEvent> {
  const reader = stream.getReader();
  const decoder = new TextDecoder();
  let buffer = '';

  try {
    for (;;) {
      const { value, done } = await reader.read();
      if (done) break;
      buffer += decoder.decode(value, { stream: true });

      let sep = buffer.indexOf('\n\n');
      while (sep !== -1) {
        const chunk = buffer.slice(0, sep);
        buffer = buffer.slice(sep + 2);
        const parsed = parseChunk(chunk);
        if (parsed) yield parsed;
        sep = buffer.indexOf('\n\n');
      }
    }
    const rest = parseChunk(buffer);
    if (rest) yield rest;
  } finally {
    reader.releaseLock();
  }
}

function parseChunk(chunk: string): SSEEvent | null {
  const trimmed = chunk.replace(/\r/g, '').trim();
  if (!trimmed) return null;
  let event = 'message';
  const dataLines: string[] = [];
  for (const line of trimmed.split('\n')) {
    if (line.startsWith(':')) continue;
    if (line.startsWith('event:')) event = line.slice(6).trim();
    else if (line.startsWith('data:')) dataLines.push(line.slice(5).trim());
  }
  if (!dataLines.length) return null;
  return { event, data: dataLines.join('\n') };
}

/** 上流のステータスコードを、内部情報を含まないユーザー向け文言へ変換する */
export function friendlyUpstreamError(status: number): string {
  if (status === 400) return 'リクエストの内容が受け付けられませんでした。入力を短くして再試行してください。';
  if (status === 401 || status === 403) return 'サーバー側の設定に問題があります。管理者にお問い合わせください。';
  if (status === 404) return '指定されたモデルを利用できません。別のモデルを選択してください。';
  if (status === 413) return '入力が長すぎます。会話を新しく開始するか、内容を短くしてください。';
  if (status === 429) return '現在混み合っています。しばらく待ってから再試行してください。';
  if (status === 529) return 'サービスが一時的に混雑しています。時間をおいて再試行してください。';
  if (status >= 500) return '応答の生成に失敗しました。時間をおいて再試行してください。';
  return '応答の生成に失敗しました。時間をおいて再試行してください。';
}

// File: src/index.ts
import {
  SESSION_COOKIE,
  buildSessionCookie,
  clearSessionCookie,
  createSessionToken,
  parseCookies,
  timingSafeEqual,
  verifySessionToken,
} from './auth';
import {
  EFFORT_LEVELS,
  MODELS,
  buildRequestBody,
  findEffort,
  findModel,
  type ChatMessage,
} from './models';
import { callAnthropic, friendlyUpstreamError, parseSSE } from './anthropic';

export interface Env {
  ANTHROPIC_API_KEY?: string;
  ACCESS_CODE?: string;
  SESSION_SECRET?: string;
  ASSETS: { fetch: (request: Request) => Promise<Response> };
}

const MAX_MESSAGES = 40;
const MAX_TOTAL_CHARS = 60_000;
const DEFAULT_ACCESS_CODE = '1359';

function json(data: unknown, status = 200, extraHeaders: Record<string, string> = {}): Response {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      'content-type': 'application/json; charset=utf-8',
      'cache-control': 'no-store',
      ...extraHeaders,
    },
  });
}

function isSecure(url: URL): boolean {
  return url.protocol === 'https:';
}

async function hasValidSession(request: Request, env: Env): Promise<boolean> {
  if (!env.SESSION_SECRET) return false;
  const cookies = parseCookies(request.headers.get('cookie'));
  return verifySessionToken(cookies[SESSION_COOKIE], env.SESSION_SECRET);
}

type ValidationResult =
  | { ok: true; modelId: string; effortId: string; messages: ChatMessage[] }
  | { ok: false; error: string };

export function validateChatPayload(payload: unknown): ValidationResult {
  if (typeof payload !== 'object' || payload === null) {
    return { ok: false, error: 'リクエストの形式が正しくありません。' };
  }
  const body = payload as Record<string, unknown>;

  if (!findModel(body.model)) {
    return { ok: false, error: '選択されたモデルは利用できません。' };
  }
  const effortId = typeof body.effort === 'string' ? body.effort : 'off';
  if (!findEffort(effortId)) {
    return { ok: false, error: '推論量の指定が正しくありません。' };
  }
  if (!Array.isArray(body.messages) || body.messages.length === 0) {
    return { ok: false, error: 'メッセージが空です。' };
  }
  if (body.messages.length > MAX_MESSAGES) {
    return { ok: false, error: '会話が長くなりすぎました。新しい会話を開始してください。' };
  }

  const messages: ChatMessage[] = [];
  let total = 0;
  for (const raw of body.messages) {
    if (typeof raw !== 'object' || raw === null) {
      return { ok: false, error: 'メッセージの形式が正しくありません。' };
    }
    const m = raw as Record<string, unknown>;
    if (m.role !== 'user' && m.role !== 'assistant') {
      return { ok: false, error: 'メッセージの形式が正しくありません。' };
    }
    if (typeof m.content !== 'string' || m.content.trim() === '') {
      return { ok: false, error: '空のメッセージは送信できません。' };
    }
    total += m.content.length;
    messages.push({ role: m.role, content: m.content });
  }
  if (total > MAX_TOTAL_CHARS) {
    return { ok: false, error: '入力が長すぎます。会話を新しく開始するか、内容を短くしてください。' };
  }
  if (messages[messages.length - 1].role !== 'user') {
    return { ok: false, error: 'メッセージの順序が正しくありません。' };
  }

  return { ok: true, modelId: body.model as string, effortId, messages };
}

/** 上流 SSE を、クライアント向けの最小限のイベントへ変換する */
export function toClientStream(upstream: ReadableStream<Uint8Array>): ReadableStream<Uint8Array> {
  const enc = new TextEncoder();
  return new ReadableStream<Uint8Array>({
    async start(controller) {
      const send = (obj: unknown) => controller.enqueue(enc.encode(`data: ${JSON.stringify(obj)}\n\n`));
      let stopReason: string | null = null;
      let sentDone = false;

      try {
        for await (const ev of parseSSE(upstream)) {
          if (ev.event === 'ping') continue;

          let parsed: Record<string, unknown>;
          try {
            parsed = JSON.parse(ev.data) as Record<string, unknown>;
          } catch {
            continue;
          }

          switch (parsed.type) {
            case 'content_block_delta': {
              const delta = parsed.delta as Record<string, unknown> | undefined;
              if (delta?.type === 'text_delta' && typeof delta.text === 'string') {
                send({ type: 'text', text: delta.text });
              } else if (delta?.type === 'thinking_delta' && typeof delta.thinking === 'string') {
                send({ type: 'thinking', text: delta.thinking });
              }
              break;
            }
            case 'message_delta': {
              const delta = parsed.delta as Record<string, unknown> | undefined;
              if (typeof delta?.stop_reason === 'string') stopReason = delta.stop_reason;
              break;
            }
            case 'message_stop': {
              send({ type: 'done', stopReason });
              sentDone = true;
              break;
            }
            case 'error': {
              console.error('anthropic stream error', JSON.stringify(parsed).slice(0, 500));
              send({ type: 'error', message: '応答の生成中にエラーが発生しました。' });
              sentDone = true;
              break;
            }
            default:
              break;
          }
        }
        if (!sentDone) send({ type: 'done', stopReason });
      } catch (err) {
        console.error('stream relay failed', err);
        try {
          send({ type: 'error', message: '通信が中断されました。再試行してください。' });
        } catch {
          /* controller already closed */
        }
      } finally {
        controller.close();
      }
    },
  });
}

async function handleLogin(request: Request, env: Env, url: URL): Promise<Response> {
  if (!env.SESSION_SECRET) {
    console.error('SESSION_SECRET is not configured');
    return json({ error: 'サーバーの設定が完了していません。管理者にお問い合わせください。' }, 500);
  }

  let body: { code?: unknown };
  try {
    body = (await request.json()) as { code?: unknown };
  } catch {
    return json({ error: 'リクエストの形式が正しくありません。' }, 400);
  }

  const input = typeof body.code === 'string' ? body.code.trim() : '';
  const expected = env.ACCESS_CODE || DEFAULT_ACCESS_CODE;

  if (!input || !timingSafeEqual(input, expected)) {
    // 総当たりへの気休め程度の緩和
    await new Promise((r) => setTimeout(r, 400));
    return json({ error: 'アクセスコードが正しくありません。' }, 401);
  }

  const token = await createSessionToken(env.SESSION_SECRET);
  return json({ ok: true }, 200, { 'set-cookie': buildSessionCookie(token, isSecure(url)) });
}

async function handleChat(request: Request, env: Env): Promise<Response> {
  if (!env.ANTHROPIC_API_KEY) {
    console.error('ANTHROPIC_API_KEY is not configured');
    return json({ error: 'サーバーの設定が完了していません。管理者にお問い合わせください。' }, 500);
  }

  let payload: unknown;
  try {
    payload = await request.json();
  } catch {
    return json({ error: 'リクエストの形式が正しくありません。' }, 400);
  }

  const validated = validateChatPayload(payload);
  if (!validated.ok) return json({ error: validated.error }, 400);

  const model = findModel(validated.modelId)!;
  const effort = findEffort(validated.effortId)!;
  const requestBody = buildRequestBody({ model, effort, messages: validated.messages });

  let upstream: Response;
  try {
    upstream = await callAnthropic(env.ANTHROPIC_API_KEY, requestBody);
  } catch (err) {
    console.error('upstream fetch failed', err);
    return json({ error: 'Claude API に接続できませんでした。時間をおいて再試行してください。' }, 502);
  }

  if (!upstream.ok || !upstream.body) {
    const detail = await upstream.text().catch(() => '');
    console.error('upstream error', upstream.status, detail.slice(0, 500));
    return json({ error: friendlyUpstreamError(upstream.status) }, upstream.status === 429 ? 429 : 502);
  }

  return new Response(toClientStream(upstream.body), {
    headers: {
      'content-type': 'text/event-stream; charset=utf-8',
      'cache-control': 'no-store, no-transform',
      'x-accel-buffering': 'no',
    },
  });
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const url = new URL(request.url);

    if (!url.pathname.startsWith('/api/')) {
      return env.ASSETS.fetch(request);
    }

    try {
      if (url.pathname === '/api/login' && request.method === 'POST') {
        return await handleLogin(request, env, url);
      }

      if (url.pathname === '/api/logout' && request.method === 'POST') {
        return json({ ok: true }, 200, { 'set-cookie': clearSessionCookie(isSecure(url)) });
      }

      if (url.pathname === '/api/session' && request.method === 'GET') {
        return json({ authenticated: await hasValidSession(request, env) });
      }

      const authed = await hasValidSession(request, env);
      if (!authed) {
        return json({ error: '認証が必要です。もう一度ログインしてください。' }, 401);
      }

      if (url.pathname === '/api/models' && request.method === 'GET') {
        return json({
          models: MODELS.map(({ id, label, description, supportsThinking }) => ({
            id,
            label,
            description,
            supportsThinking,
          })),
          efforts: EFFORT_LEVELS.map(({ id, label }) => ({ id, label })),
        });
      }

      if (url.pathname === '/api/chat' && request.method === 'POST') {
        return await handleChat(request, env);
      }

      return json({ error: '要求されたリソースが見つかりません。' }, 404);
    } catch (err) {
      console.error('unhandled error', err);

// File: public/markdown.js
/**
 * 依存関係なしの軽量 Markdown レンダラ。
 * すべての入力を先に HTML エスケープしてから構造を組み立てるため、
 * 生の HTML が出力に混入することはない（XSS 対策）。
 */

const ESCAPE_MAP = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' };

export function escapeHtml(input) {
  return String(input ?? '').replace(/[&<>"']/g, (ch) => ESCAPE_MAP[ch]);
}

function safeUrl(raw) {
  const url = String(raw).trim();
  if (/^(?:https?:\/\/|mailto:)/i.test(url)) return url;
  if (/^[#/]/.test(url)) return url;
  return null;
}

function applyEmphasis(text) {
  return text
    .replace(/\*\*([\s\S]+?)\*\*/g, '<strong>$1</strong>')
    .replace(/~~([\s\S]+?)~~/g, '<del>$1</del>')
    .replace(/\*([^\s*][\s\S]*?)\*/g, '<em>$1</em>');
}

export function renderInline(text, options = {}) {
  const breaks = options.breaks === true;
  const codeSpans = [];
  const links = [];

  // 1. コードスパンを退避（内部は後続の変換対象外）
  let out = String(text ?? '').replace(/(`+)([\s\S]+?)\1/g, (_m, _ticks, body) => {
    codeSpans.push(escapeHtml(body.replace(/^ (.*) $/, '$1')));
    return `\u0000C${codeSpans.length - 1}\u0000`;
  });

  // 2. エスケープ
  out = escapeHtml(out);

  // 3. リンクを退避
  out = out.replace(/!?\[([^\]]*)\]\(([^)\s]+)(?:\s+&quot;[^)]*&quot;)?\)/g, (match, label, href) => {
    const url = safeUrl(href.replace(/&amp;/g, '&'));
    if (!url) return match;
    links.push({ href: escapeHtml(url), label });
    return `\u0000L${links.length - 1}\u0000`;
  });

  // 4. 強調
  out = applyEmphasis(out);

  // 5. 改行
  if (breaks) out = out.replace(/\n/g, '<br>');

  // 6. 復元
  out = out.replace(/\u0000L(\d+)\u0000/g, (_m, i) => {
    const link = links[Number(i)];
    return `<a href="${link.href}" target="_blank" rel="noopener noreferrer">${applyEmphasis(link.label)}</a>`;
  });
  out = out.replace(/\u0000C(\d+)\u0000/g, (_m, i) => `<code>${codeSpans[Number(i)]}</code>`);

  return out;
}

const RE_FENCE = /^ {0,3}(`{3,}|~{3,})\s*([^\s`]*)\s*$/;
const RE_HEADING = /^ {0,3}(#{1,6})\s+(.*?)\s*#*\s*$/;
const RE_HR = /^ {0,3}([-*_])\s*(?:\1\s*){2,}$/;
const RE_QUOTE = /^ {0,3}>\s?/;
const RE_LIST = /^( *)([-*+]|\d{1,9}[.)])\s+(.*)$/;

function startsNewBlock(line) {
  return (
    !line.trim() ||
    RE_FENCE.test(line) ||
    RE_HEADING.test(line) ||
    RE_HR.test(line) ||
    RE_QUOTE.test(line) ||
    RE_LIST.test(line)
  );
}

function isTableSeparator(line) {
  return /^\s*\|?[\s:|-]*-[\s:|-]*\|?\s*$/.test(line) && line.includes('-') && line.includes('|');
}

function splitRow(line) {
  return line
    .replace(/^\s*\|/, '')
    .replace(/\|\s*$/, '')
    .split('|')
    .map((cell) => cell.trim());
}

export function renderMarkdown(source) {
  const lines = String(source ?? '')
    .replace(/\r\n?/g, '\n')
    .split('\n');
  return parseBlocks(lines);
}

function parseBlocks(lines) {
  const html = [];
  let i = 0;

  while (i < lines.length) {
    const line = lines[i];

    if (!line.trim()) {
      i++;
      continue;
    }

    // фenced code block
    const fence = line.match(RE_FENCE);
    if (fence) {
      const marker = fence[1][0];
      const len = fence[1].length;
      const lang = fence[2] || '';
      const closing = new RegExp(`^ {0,3}\\${marker}{${len},}\\s*$`);
      const body = [];
      i++;
      while (i < lines.length && !closing.test(lines[i])) {
        body.push(lines[i]);
        i++;
      }
      i++;
      const cls = /^[\w+.#-]+$/.test(lang) ? ` class="language-${escapeHtml(lang)}"` : '';
      const label = /^[\w+.#-]+$/.test(lang) ? ` data-lang="${escapeHtml(lang)}"` : '';
      html.push(`<pre${label}><code${cls}>${escapeHtml(body.join('\n'))}</code></pre>`);
      continue;
    }

    // heading
    const heading = line.match(RE_HEADING);
    if (heading) {
      const level = heading[1].length;
      html.push(`<h${level}>${renderInline(heading[2])}</h${level}>`);
      i++;
      continue;
    }

    // horizontal rule
    if (RE_HR.test(line)) {
      html.push('<hr>');
      i++;
      continue;
    }

    // blockquote
    if (RE_QUOTE.test(line)) {
      const buf = [];
      while (i < lines.length && (RE_QUOTE.test(lines[i]) || (lines[i].trim() && buf.length))) {
        if (!RE_QUOTE.test(lines[i]) && !lines[i].trim()) break;
        buf.push(lines[i].replace(RE_QUOTE, ''));
        i++;
      }
      html.push(`<blockquote>${parseBlocks(buf)}</blockquote>`);
      continue;
    }

    // table
    if (line.includes('|') && i + 1 < lines.length && isTableSeparator(lines[i + 1])) {
      const header = splitRow(line);
      const aligns = splitRow(lines[i + 1]).map((cell) => {
        const left = cell.startsWith(':');
        const right = cell.endsWith(':');
        if (left && right) return ' style="text-align:center"';
        if (right) return ' style="text-align:right"';
        if (left) return ' style="text-align:left"';
        return '';
      });
      i += 2;
      const rows = [];
      while (i < lines.length && lines[i].includes('|') && lines[i].trim()) {
        rows.push(splitRow(lines[i]));
        i++;
      }
      const thead = header.map((c, idx) => `<th${aligns[idx] || ''}>${renderInline(c)}</th>`).join('');
      const tbody = rows
        .map(
          (row) =>
            `<tr>${header
              .map((_h, idx) => `<td${aligns[idx] || ''}>${renderInline(row[idx] ?? '')}</td>`)
              .join('')}</tr>`,
        )
        .join('');
      html.push(`<div class="table-wrap"><table><thead><tr>${thead}</tr></thead><tbody>${tbody}</tbody></table></div>`);
      continue;
    }

    // list
    const listStart = line.match(RE_LIST);
    if (listStart) {
      const baseIndent = listStart[1].length;
      const ordered = /^\d/.test(listStart[2]);
      const items = [];
      let current = null;

      while (i < lines.length) {
        const l = lines[i];

        if (!l.trim()) {
          const next = lines[i + 1];
          if (next && (/^ {2,}\S/.test(next) || RE_LIST.test(next))) {
            if (current) current.lines.push('');
            i++;
            continue;
          }
          break;
        }

        const m = l.match(RE_LIST);
        if (m && m[1].length <= baseIndent + 1) {
          current = { lines: [m[3]], loose: false };
          items.push(current);
          i++;
          continue;
        }
        if (current && (m || /^ {2,}\S/.test(l))) {
          current.lines.push(l.replace(new RegExp(`^ {0,${baseIndent + 2}}`), ''));
          i++;
          continue;
        }
        break;
      }

      const rendered = items
        .map((item) => {
          const tight = !item.lines.some((l) => l.trim() === '');
          let inner = parseBlocks(item.lines);
          if (tight) inner = inner.replace(/^<p>([\s\S]*?)<\/p>/, '$1');
          return `<li>${inner}</li>`;
        })
        .join('');

      const startAttr =
        ordered && listStart[2] !== '1.' && listStart[2] !== '1)'
          ? ` start="${parseInt(listStart[2], 10)}"`
          : '';
      html.push(ordered ? `<ol${startAttr}>${rendered}</ol>` : `<ul>${rendered}</ul>`);
      continue;
    }

    // paragraph
    const buf = [line];
    i++;
    while (i < lines.length && !startsNewBlock(lines[i])) {
      buf.push(lines[i]);
      i++;
    }
    html.push(`<p>${renderInline(buf.join('\n'), { breaks: true })}</p>`);
  }

  return html.join('\n');
}

async function sendMessage(text) {
  clearError();
  stickToBottom = true;

  const history = state.turns
    .filter((turn) => turn.content.trim() !== '')
    .map((turn) => ({ role: turn.role, content: turn.content }));
  history.push({ role: 'user', content: text });

  state.turns.push({ role: 'user', content: text });
  const assistant = { role: 'assistant', content: '', thinking: '', pending: true };
  state.turns.push(assistant);
  renderMessages();
  autoScroll();

  const controller = new AbortController();
  state.controller = controller;
  setStreaming(true);

  try {
    const res = await fetch('/api/chat', {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      credentials: 'same-origin',
      signal: controller.signal,
      body: JSON.stringify({
        model: els.modelSelect.value,
        effort: els.effortControl.hidden ? 'off' : els.effortSelect.value,
        messages: history,
      }),
    });

    if (res.status === 401) {
      enterLogin();
      throw new Error('セッションの有効期限が切れました。もう一度ログインしてください。');
    }

    if (!res.ok || !res.body) {
      const data = await res.json().catch(() => ({}));
      throw new Error(data.error || '応答の取得に失敗しました。');
    }

    const reader = res.body.getReader();
    const decoder = new TextDecoder();
    let buffer = '';

    for (;;) {
      const { value, done } = await reader.read();
      if (done) break;
      buffer += decoder.decode(value, { stream: true });

      let sep = buffer.indexOf('\n\n');
      while (sep !== -1) {
        const chunk = buffer.slice(0, sep);
        buffer = buffer.slice(sep + 2);
        sep = buffer.indexOf('\n\n');

        for (const line of chunk.split('\n')) {
          if (!line.startsWith('data:')) continue;
          let payload;
          try {
            payload = JSON.parse(line.slice(5).trim());
          } catch {
            continue;
          }
          if (payload.type === 'text') {
            assistant.content += payload.text;
            scheduleStreamRender();
          } else if (payload.type === 'thinking') {
            assistant.thinking += payload.text;
            scheduleStreamRender();
          } else if (payload.type === 'error') {
            showError(payload.message || '応答の生成中にエラーが発生しました。');
          }
        }
      }
    }
  } catch (err) {
    if (err.name === 'AbortError') {
      if (!assistant.content) assistant.content = '_（生成を停止しました）_';
    } else {
      showError(err.message || '通信エラーが発生しました。');
      if (!assistant.content) {
        state.turns = state.turns.filter((turn) => turn !== assistant);
      }
    }
  } finally {
    assistant.pending = false;
    state.controller = null;
    setStreaming(false);
    renderMessages();
    autoScroll();
    els.input.focus();
  }
}

<!-- File: README.md -->
# Claude Chat (Cloudflare Workers)

Claude API を利用する最小構成のチャット Web アプリ。API キーは Cloudflare Workers 側の
シークレットとしてのみ保持し、ブラウザには一切渡さない。

## 機能

- 簡易アクセスコード認証（HttpOnly + 署名付き Cookie セッション）
- モデル選択（Haiku / Sonnet / Opus）
- 推論量（effort）選択 = Extended Thinking の `budget_tokens`
- SSE によるストリーミング表示
- 入力・出力とも Markdown 対応（コードブロック、リスト、表、引用など）
- 各回答・各コードブロック・入力欄のワンクリックコピー
- ライト/ダーク対応、モバイル・デスクトップ両対応のレイアウト

## 構成

```
src/index.ts       Worker 本体（ルーティング、認証、SSE 中継）
src/auth.ts        HMAC 署名セッション
src/models.ts      モデル定義・リクエストボディ組み立て
src/anthropic.ts   Anthropic Messages API 呼び出し・SSE パース
public/            静的アセット（HTML / CSS / JS / Markdown レンダラ）
test/              Vitest によるユニットテスト
```

サーバー側ルート:

| メソッド | パス | 説明 |
| --- | --- | --- |

| POST | `/api/logout` | セッション破棄 |
| GET | `/api/session` | ログイン状態の確認 |

## 環境変数

| `ANTHROPIC_API_KEY` | Secret | Anthropic Console で発行した API キー |

## モデル ID と effort の根拠

参照する一次情報は Anthropic 公式ドキュメントのみ。

- Models overview: `docs.anthropic.com/en/docs/about-claude/models/overview`
- Messages API: `docs.anthropic.com/en/api/messages`
- Extended thinking: `docs.anthropic.com/en/docs/build-with-claude/extended-thinking`
- Streaming Messages: `docs.anthropic.com/en/docs/build-with-claude/streaming`

注意点:

- 「Sonnet 5 / Opus 5」という名称のモデルは公式の Models overview に存在しないため、
  実在する最新世代（Claude 4.5 系）の ID に置き換えている。存在しないモデル名は送信しない。
- 新しい世代が公開された場合は `src/models.ts` の `MODELS` にある `id` を、
  公式 Models overview に記載された Model ID へ更新するだけでよい。
  UI・API・テストは自動的に追従する（テストは ID の重複と接頭辞のみ検証する）。
- 「推論量（effort）」は、公式に仕様が公開されている Extended Thinking の
  `thinking.budget_tokens` へマッピングしている。仕様が確認できない非公開パラメータは送信しない。
  将来 `effort` パラメータが公式ドキュメントに追加された場合は、
  `src/models.ts` の `buildRequestBody()` のみを差し替えれば対応できる。
- Extended Thinking 有効時は `max_tokens > budget_tokens` を満たし、`temperature` を送らない
  （公式仕様に準拠）。

## セキュリティ上の設計と制限

- API キーは Worker のシークレットにのみ存在し、レスポンスにもクライアントコードにも含まれない。
- 上流エラーは `console.error` にのみ記録し、画面には汎用の日本語メッセージだけを表示する。
- セッションは HMAC-SHA256 署名付きトークンを HttpOnly / SameSite=Lax Cookie に格納する
  （有効期限 12 時間）。
- Markdown はサニタイズ済みの自前レンダラで描画する（生 HTML は出力しない）。
- アクセスコード認証はあくまで簡易的なアクセス制限であり、本格的な認証基盤ではない。
  失敗時に短い遅延を入れているが、厳密なレート制限は行っていない。
  より強い保護が必要な場合は Cloudflare Access 等を前段に置くこと。
- 会話履歴はブラウザのメモリ上のみに保持し、サーバーには保存しない
  （リロードで消える。モデルと effort の選択のみ `localStorage` に保存）。
