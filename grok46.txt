// File: src/auth.ts
import type { Env } from "./types";

const COOKIE_NAME = "session";
const SESSION_TTL_MS = 7 * 24 * 60 * 60 * 1000;

export function expectedAccessCode(env: Env): string {
  return env.ACCESS_CODE && env.ACCESS_CODE.length > 0 ? env.ACCESS_CODE : "1359";
}

export function timingSafeEqual(a: string, b: string): boolean {
  const encoder = new TextEncoder();
  const aa = encoder.encode(a);
  const bb = encoder.encode(b);
  const len = Math.max(aa.length, bb.length);
  let out = aa.length === bb.length ? 0 : 1;
  for (let i = 0; i < len; i += 1) {
    const av = i < aa.length ? aa[i] : 0;
    const bv = i < bb.length ? bb[i] : 0;
    out |= av ^ bv;
  }
  return out === 0;
}

export async function hmacHex(secret: string, data: string): Promise<string> {
  const encoder = new TextEncoder();
  const key = await crypto.subtle.importKey(
    "raw",
    encoder.encode(secret),
    { name: "HMAC", hash: "SHA-256" },
    false,
    ["sign"],
  );
  const sig = await crypto.subtle.sign("HMAC", key, encoder.encode(data));
  return Array.from(new Uint8Array(sig), (byte) => byte.toString(16).padStart(2, "0")).join("");
}

export async function createSessionToken(secret: string, now = Date.now()): Promise<string> {
  const payload = String(now + SESSION_TTL_MS);
  const sig = await hmacHex(secret, payload);
  return `${payload}.${sig}`;
}

export async function verifySessionToken(
  secret: string,
  token: string,
  now = Date.now(),
): Promise<boolean> {
  const index = token.lastIndexOf(".");
  if (index <= 0) return false;
  const payload = token.slice(0, index);
  const sig = token.slice(index + 1);
  const expected = await hmacHex(secret, payload);
  if (!timingSafeEqual(sig, expected)) return false;
  const exp = Number(payload);
  return Number.isFinite(exp) && exp >= now;
}

export function readCookie(header: string | null, name: string): string | null {
  if (!header) return null;
  const parts = header.split(";");
  for (const part of parts) {
    const [key, ...rest] = part.trim().split("=");
    if (key === name) {
      try {
        return decodeURIComponent(rest.join("="));
      } catch {
        return rest.join("=");
      }
    }
  }
  return null;
}

export function buildSessionCookie(token: string, request: Request): string {
  const secure = new URL(request.url).protocol === "https:" ? "; Secure" : "";
  const maxAge = Math.floor(SESSION_TTL_MS / 1000);
  return `${COOKIE_NAME}=${encodeURIComponent(token)}; HttpOnly; Path=/; SameSite=Lax; Max-Age=${maxAge}${secure}`;
}

export function clearSessionCookie(request: Request): string {
  const secure = new URL(request.url).protocol === "https:" ? "; Secure" : "";
  return `${COOKIE_NAME}=; HttpOnly; Path=/; SameSite=Lax; Max-Age=0${secure}`;
}

export async function isAuthenticated(request: Request, env: Env): Promise<boolean> {
  const secret = env.SESSION_SECRET ?? "";
  if (!secret) return false;
  const token = readCookie(request.headers.get("Cookie"), COOKIE_NAME);
  if (!token) return false;
  return verifySessionToken(secret, token);
}

// File: src/claude.ts
import { publicUpstreamError } from "./errors";
import type { ChatMessage, EffortDef, ModelDef } from "./types";

const ANTHROPIC_URL = "https://api.anthropic.com/v1/messages";
const ANTHROPIC_VERSION = "2023-06-01";
const MAX_MESSAGES = 40;
const MAX_CONTENT_CHARS = 100_000;

export function normalizeMessages(input: unknown): ChatMessage[] | null {
  if (!Array.isArray(input)) return null;
  const out: ChatMessage[] = [];
  for (const item of input) {
    if (!item || typeof item !== "object") return null;
    const role = (item as { role?: unknown }).role;
    const content = (item as { content?: unknown }).content;
    if (role !== "user" && role !== "assistant") return null;
    if (typeof content !== "string") return null;
    const trimmed = content.slice(0, MAX_CONTENT_CHARS);
    if (trimmed.length === 0) continue;
    out.push({ role, content: trimmed });
  }
  if (out.length === 0) return null;
  const clipped = out.slice(-MAX_MESSAGES);
  if (clipped[clipped.length - 1].role !== "user") return null;
  return clipped;
}

export function buildClaudeBody(
  model: ModelDef,
  effort: EffortDef,
  messages: ChatMessage[],
): Record<string, unknown> {
  const maxTokens = Math.min(Math.max(effort.maxTokens, 256), model.maxOutputTokens);
  const body: Record<string, unknown> = {
    model: model.id,
    max_tokens: maxTokens,
    stream: true,
    messages,
  };
  if (effort.budget > 0 && effort.budget < maxTokens) {
    body.thinking = {
      type: "enabled",
      budget_tokens: effort.budget,
    };
  }
  return body;
}

export function withoutThinking(body: Record<string, unknown>): Record<string, unknown> {
  const next = { ...body };
  delete next.thinking;
  return next;
}

export function consumeSseBlock(block: string): unknown | undefined {
  const data = block
    .replace(/\r\n/g, "\n")
    .split("\n")
    .filter((line) => line.startsWith("data:"))
    .map((line) => line.slice(5).trimStart())
    .join("");
  if (!data || data === "[DONE]") return undefined;
  try {
    return JSON.parse(data);
  } catch {
    return undefined;
  }
}

export function mapClaudeEvent(
  evt: unknown,
): { type: "text"; text: string } | { type: "done" } | { type: "error"; message: string } | null {
  if (!evt || typeof evt !== "object") return null;
  const type = (evt as { type?: unknown }).type;
  if (type === "content_block_delta") {
    const delta = (evt as { delta?: { type?: unknown; text?: unknown } }).delta;
    if (delta?.type === "text_delta" && typeof delta.text === "string" && delta.text.length > 0) {
      return { type: "text", text: delta.text };
    }
    return null;
  }
  if (type === "message_stop") {
    return { type: "done" };
  }
  if (type === "error") {
    return { type: "error", message: publicUpstreamError(500) };
  }
  return null;
}

function encodeEvent(obj: unknown): Uint8Array {
  return new TextEncoder().encode(`data: ${JSON.stringify(obj)}\n\n`);
}

export function createClaudeToClientTransform(): TransformStream<Uint8Array, Uint8Array> {
  let buffer = "";
  let sentDone = false;
  const decoder = new TextDecoder();

  return new TransformStream({
    transform(chunk, controller) {
      buffer += decoder.decode(chunk, { stream: true });
      buffer = buffer.replace(/\r\n/g, "\n");
      const parts = buffer.split("\n\n");
      buffer = parts.pop() ?? "";
      for (const part of parts) {
        const evt = consumeSseBlock(part);
        const mapped = mapClaudeEvent(evt);
        if (!mapped) continue;
        controller.enqueue(encodeEvent(mapped));
        if (mapped.type === "done" || mapped.type === "error") {
          sentDone = true;
        }
      }
    },
    flush(controller) {
      if (buffer.trim()) {
        const evt = consumeSseBlock(buffer);
        const mapped = mapClaudeEvent(evt);
        if (mapped) {
          controller.enqueue(encodeEvent(mapped));
          if (mapped.type === "done" || mapped.type === "error") {
            sentDone = true;
          }
        }
      }
      if (!sentDone) {
        controller.enqueue(encodeEvent({ type: "done" }));
      }
    },
  });
}

export async function streamClaude(options: {
  apiKey: string;
  body: Record<string, unknown>;
  signal?: AbortSignal;
  fetchImpl?: typeof fetch;
}): Promise<Response> {
  const fetchImpl = options.fetchImpl ?? fetch;
  const headers = {
    "content-type": "application/json",
    "x-api-key": options.apiKey,
    "anthropic-version": ANTHROPIC_VERSION,
  };

  let res = await fetchImpl(ANTHROPIC_URL, {
    method: "POST",
    headers,
    body: JSON.stringify(options.body),
    signal: options.signal,
  });

  if (res.status === 400 && options.body.thinking) {
    if (res.body) {
      await res.body.cancel();
    }
    res = await fetchImpl(ANTHROPIC_URL, {
      method: "POST",
      headers,
      body: JSON.stringify(withoutThinking(options.body)),
      signal: options.signal,
    });
  }

  return res;
}

import {
  buildSessionCookie,

  expectedAccessCode,
  isAuthenticated,

import { publicAppError, publicUpstreamError } from "./errors";
import { getEffort, getModel, publicConfig } from "./models";
import type { Env } from "./types";

function json(data: unknown, status = 200): Response {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      "Content-Type": "application/json; charset=utf-8",
      "Cache-Control": "no-store",
    },
  });
}

async function readJson(request: Request): Promise<unknown | null> {
  try {
    return await request.json();
  } catch {
    return null;
  }
}

async function requireAuth(request: Request, env: Env): Promise<Response | null> {
  if (await isAuthenticated(request, env)) return null;
  return json({ error: "ログインしてください。" }, 401);
}

async function handleLogin(request: Request, env: Env): Promise<Response> {
  if (request.method !== "POST") {
    return json({ error: "リクエストを処理できませんでした。" }, 405);
  }
  if (!env.SESSION_SECRET) {
    return json({ error: "サーバーの設定が完了していません。" }, 500);
  }
  const body = await readJson(request);
  const code =
    body && typeof body === "object" && "code" in body
      ? String((body as { code?: unknown }).code ?? "").trim()
      : "";
  if (!timingSafeEqual(code, expectedAccessCode(env))) {
    return json({ error: "コードが違います。" }, 401);
  }
  const token = await createSessionToken(env.SESSION_SECRET);
  const res = json({ ok: true });
  res.headers.set("Set-Cookie", buildSessionCookie(token, request));
  return res;
}

async function handleLogout(request: Request): Promise<Response> {
  if (request.method !== "POST") {
    return json({ error: "リクエストを処理できませんでした。" }, 405);
  }
  const res = json({ ok: true });
  res.headers.set("Set-Cookie", clearSessionCookie(request));
  return res;
}

async function handleSession(request: Request, env: Env): Promise<Response> {
  if (request.method !== "GET") {
    return json({ error: "リクエストを処理できませんでした。" }, 405);
  }
  const ok = await isAuthenticated(request, env);
  return json({ ok }, ok ? 200 : 401);
}

async function handleConfig(request: Request, env: Env): Promise<Response> {
  if (request.method !== "GET") {
    return json({ error: "リクエストを処理できませんでした。" }, 405);
  }
  const unauthorized = await requireAuth(request, env);
  if (unauthorized) return unauthorized;
  return json(publicConfig());
}

async function handleChat(request: Request, env: Env): Promise<Response> {
  if (request.method !== "POST") {
    return json({ error: "リクエストを処理できませんでした。" }, 405);
  }
  const unauthorized = await requireAuth(request, env);
  if (unauthorized) return unauthorized;
  if (!env.ANTHROPIC_API_KEY) {
    return json({ error: "サーバーの設定が完了していません。" }, 500);
  }

  const body = await readJson(request);
  if (!body || typeof body !== "object") {
    return json({ error: "リクエストを処理できませんでした。" }, 400);
  }

  const model = getModel((body as { model?: unknown }).model);
  if (!model) {
    return json({ error: "リクエストを処理できませんでした。" }, 400);
  }
  const effort = getEffort((body as { effort?: unknown }).effort);
  const messages = normalizeMessages((body as { messages?: unknown }).messages);
  if (!messages) {
    return json({ error: "リクエストを処理できませんでした。" }, 400);
  }

  const claudeBody = buildClaudeBody(model, effort, messages);
  const upstream = await streamClaude({
    apiKey: env.ANTHROPIC_API_KEY,
    body: claudeBody,
    signal: request.signal,
  });

  if (!upstream.ok || !upstream.body) {
    console.error("claude_error", upstream.status);
    const status = upstream.status === 401 || upstream.status === 403 ? 500 : 502;
    return json({ error: publicUpstreamError(upstream.status) }, status);
  }

  return new Response(upstream.body.pipeThrough(createClaudeToClientTransform()), {
    status: 200,
    headers: {
      "Content-Type": "text/event-stream; charset=utf-8",
      "Cache-Control": "no-store",
    },
  });
}

async function handleApi(request: Request, env: Env, path: string): Promise<Response> {
  if (path === "/api/login") return handleLogin(request, env);
  if (path === "/api/logout") return handleLogout(request);
  if (path === "/api/session") return handleSession(request, env);
  if (path === "/api/config") return handleConfig(request, env);
  if (path === "/api/chat") return handleChat(request, env);
  return json({ error: "見つかりません。" }, 404);
}

export async function handleRequest(request: Request, env: Env): Promise<Response> {
  const url = new URL(request.url);
  try {
    if (url.pathname.startsWith("/api/")) {
      return await handleApi(request, env, url.pathname);
    }
    if (!env.ASSETS) {
      return new Response("Not found", { status: 404 });
    }
    return env.ASSETS.fetch(request);
  } catch {
    console.error("unhandled");
    return json({ error: publicAppError() }, 500);
  }
}

export default {
  fetch(request: Request, env: Env): Promise<Response> {
    return handleRequest(request, env);
  },
};

function escapeHtml(value) {
  return String(value).replace(/[&<>"']/g, (ch) => {
    switch (ch) {
      case "&":
        return "&amp;";
      case "<":
        return "&lt;";
      case ">":
        return "&gt;";
      case '"':
        return "&quot;";
      default:
        return "&#39;";
    }
  });
}

function renderMarkdown(src) {
  const parsed = window.marked.parse(src || "", { gfm: true, breaks: true });
  const clean = window.DOMPurify.sanitize(parsed, { USE_PROFILES: { html: true } });
  const holder = document.createElement("div");
  holder.innerHTML = clean;
  holder.querySelectorAll("a").forEach((anchor) => {
    anchor.target = "_blank";
    anchor.rel = "noopener noreferrer";
  });
  return holder.innerHTML;
}

async function copyText(text, button) {
  const value = text ?? "";
  try {
    await navigator.clipboard.writeText(value);
  } catch {
    const area = document.createElement("textarea");

function addMessage(role, content, streaming = false) {

async function readSse(response, onEvent) {
  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let buffer = "";
  while (true) {
    const { value, done } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });
    const parts = buffer.split("\n\n");
    buffer = parts.pop() ?? "";
    for (const part of parts) {
      const line = part.split("\n").find((item) => item.startsWith("data: "));
      if (!line) continue;
      try {
        onEvent(JSON.parse(line.slice(6)));
      } catch {
        // ignore malformed chunk
      }
    }
  }
}

  event.preventDefault();
  const text = promptEl.value.trim();
  if (!text || state.streaming) return;

  showError(chatError, "");
  state.messages.push({ role: "user", content: text });
  addMessage("user", text);
  promptEl.value = "";
  resizePrompt();
  updateEmpty();

  const assistant = { role: "assistant", content: "" };
  state.messages.push(assistant);
  const article = addMessage("assistant", "", true);

  const payload = {
    model: modelSelect.value,
    effort: effortSelect.value,
    messages: state.messages
      .filter((item, index, arr) => !(index === arr.length - 1 && item.role === "assistant"))
      .map(({ role, content }) => ({ role, content })),
  };

  state.abort = new AbortController();
  setStreaming(true);

  try {
    const res = await api("/api/chat", {
      method: "POST",
      body: JSON.stringify(payload),
      signal: state.abort.signal,
      headers: { "Content-Type": "application/json" },
    });

    if (!res.ok) {
      const data = await res.json().catch(() => ({}));
      throw new Error(data.error || "エラーが発生しました。時間をおいて再試行してください。");
    }

    await readSse(res, (msg) => {
      if (msg.type === "text") {
        assistant.content += msg.text;
        updateMessage(article, assistant.content, true);
      } else if (msg.type === "error") {
        throw new Error(msg.message || "応答の生成に失敗しました。時間をおいて再試行してください。");
      }
    });
    updateMessage(article, assistant.content, false);
    if (!assistant.content) {
      showError(chatError, "応答の生成に失敗しました。時間をおいて再試行してください。");
    }
  } catch (err) {
    if (err && err.name === "AbortError") return;
    updateMessage(article, assistant.content, false);
    showError(
      chatError,
      err && err.message ? err.message : "エラーが発生しました。時間をおいて再試行してください。",
    );
  } finally {
    state.abort = null;
    setStreaming(false);
  }
});

# Claude Chat

Cloudflare Workers 上で動く、Claude API の簡易チャットです。
API キーはサーバー側の環境変数のみで扱い、ブラウザには出しません。

## 機能

- アクセスコードによる簡易ログイン（コードは `1359`）
- モデル選択（Haiku 3.5 / Sonnet 3.5 / Opus 3）
- effort 選択（公式の `thinking.budget_tokens` にマップ）
- チャット履歴表示とストリーミング応答
- 入力・出力の Markdown 表示
- 入力欄と各回答のワンクリックコピー

## モデルについて

指定の Haiku 4.5 / Sonnet 5 / Opus 5 は、Anthropic 公式ドキュメントで利用を確認できませんでした。
未提供の名前は使わず、公式のモデル ID に置き換えています。

- Haiku 3.5: `claude-3-5-haiku-20241022`
- Sonnet 3.5: `claude-3-5-sonnet-20241022`
- Opus 3: `claude-3-opus-20240229`

更新する場合は [Anthropic のモデル一覧](https://docs.anthropic.com/en/docs/about-claude/models) を確認し、`src/models.ts` を書き換えてください。

## effort について

UI の低 / 中 / 高は、公式 Messages API の extended thinking（`thinking.budget_tokens`）にマップします。
非対応モデル、または `budget_tokens` が `max_tokens` 以上になる場合は thinking を付けません。
API が 400 を返した場合は thinking なしで 1 回だけ再試行します。

## 必要要件

- Node.js 18 以降
- Anthropic API キー
- Cloudflare アカウント（デプロイ時）

## 環境変数

| 名前 | 必須 | 説明 |
| --- | --- | --- |
| `ANTHROPIC_API_KEY` | はい | Claude API キー。クライアントへは渡しません |
| `SESSION_SECRET` | はい | セッション Cookie 署名用。32 文字以上を推奨 |
| `ACCESS_CODE` | いいえ | 未設定時は `1359` |

## ローカルでの起動方法

1. 依存関係を入れる

        npm install

2. 環境変数ファイルを作る

        cp .dev.vars.example .dev.vars

3. `.dev.vars` を編集する

        ANTHROPIC_API_KEY=（発行したキー）
        SESSION_SECRET=（例: openssl rand -hex 32 の出力）
        ACCESS_CODE=1359

4. テストを実行する

        npm test

5. 開発サーバーを起動する

        npm run dev

6. ブラウザで `http://localhost:8787` を開く
7. アクセスコード `1359` を入力する（画面の文字種説明は実際の条件ではありません）

## Cloudflare へのデプロイ手順

1. Cloudflare にログインする

        npx wrangler login

2. シークレットを登録する

        npx wrangler secret put ANTHROPIC_API_KEY
        npx wrangler secret put SESSION_SECRET

3. アクセスコードを変える場合のみ

        npx wrangler secret put ACCESS_CODE

4. デプロイする

        npm run deploy

5. 表示された `*.workers.dev` URL を開き、`1359` でログインする

長文のストリーミングは Workers の実行時間制限の影響を受けます。必要なら有料プランを使ってください。

## セキュリティ

- Claude API キーは Worker の環境変数のみ
- セッションは HttpOnly Cookie（HMAC 署名）
- モデル ID はサーバー側の許可リストでのみ解決
- エラー応答から API キーと内部情報を除去

## テスト

        npm test
