## 実装判断（Claude API 仕様の一次情報に基づく部分）

- モデルID: Anthropic公式ドキュメント「Models overview」時点で `Sonnet 5` / `Opus 5` は提供されていないため、要件の指示どおり「利用可能なもの」に置き換え
  - Claude Haiku 4.5 → `claude-haiku-4-5-20251001`
  - Claude Sonnet 4.5 → `claude-sonnet-4-5-20250929`
  - Claude Opus 4.5 → `claude-opus-4-5-20251101`
  - 新モデル公開時は `src/config.js` の1箇所だけ更新すればよい構造
- effort: 公式ドキュメント「Effort」に基づき、Opus 4.5 のみ対応（`output_config.effort` + betaヘッダ `effort-2025-11-24`、値は `low` / `medium` / `high`）
  - 非対応モデルではUI側で無効化し、サーバー側でもパラメータを送信しない二重ガード
- ストリーミング: Messages API の `stream: true`（SSE）をWorkerで中継し、ブラウザで逐次レンダリング
- 認証: HMAC-SHA256署名付きHttpOnly Cookie（7日間有効）。アクセスコードは環境変数 `ACCESS_CODE`、未設定時のみ `1359` にフォールバック
- APIキー: Cloudflareのシークレット `CLAUDE_API_KEY` のみ。クライアントコード・レスポンス・エラー文に一切含めない

```js
// File: src/auth.js
import { SESSION_TTL_SECONDS } from "./config.js";

const encoder = new TextEncoder();

// 長さが異なる場合は即 false（簡易アクセス制限用途のため許容）
export function timingSafeEqual(a, b) {
  if (typeof a !== "string" || typeof b !== "string") return false;
  if (a.length === 0 || a.length !== b.length) return false;
  let diff = 0;
  for (let i = 0; i < a.length; i++) {
    diff |= a.charCodeAt(i) ^ b.charCodeAt(i);
  }
  return diff === 0;
}

async function hmacHex(secret, message) {
  const key = await crypto.subtle.importKey(
    "raw",
    encoder.encode(secret),
    { name: "HMAC", hash: "SHA-256" },
    false,
    ["sign"],
  );
  const signature = await crypto.subtle.sign("HMAC", key, encoder.encode(message));
  return Array.from(new Uint8Array(signature), (b) => b.toString(16).padStart(2, "0")).join("");
}

// セッション署名の秘密鍵。AUTH_SECRET 優先、未設定なら API キーから派生させる
export function resolveAuthSecret(env) {
  return env.AUTH_SECRET || env.CLAUDE_API_KEY || "";
}

export async function createSessionCookie(secret, ttlSeconds = SESSION_TTL_SECONDS) {
  const expires = Math.floor(Date.now() / 1000) + ttlSeconds;
  const signature = await hmacHex(secret, String(expires));
  const maxAge = Math.max(ttlSeconds, 0);
  return `session=${expires}.${signature}; HttpOnly; Secure; SameSite=Strict; Path=/; Max-Age=${maxAge}`;
}

export function clearSessionCookie() {
  return "session=; HttpOnly; Secure; SameSite=Strict; Path=/; Max-Age=0";
}

function readCookie(request, name) {
  const header = request.headers.get("Cookie") || "";
  for (const part of header.split(";")) {
    const [key, ...rest] = part.trim().split("=");
    if (key === name) return rest.join("=");
  }
  return "";
}

export async function verifyRequest(request, secret) {
  if (!secret) return false;
  const token = readCookie(request, "session");
  const dot = token.lastIndexOf(".");
  if (dot <= 0) return false;
  const expiresStr = token.slice(0, dot);
  const signature = token.slice(dot + 1);
  const expires = Number(expiresStr);
  if (!Number.isInteger(expires) || expires < Math.floor(Date.now() / 1000)) return false;
  const expected = await hmacHex(secret, expiresStr);
  return timingSafeEqual(expected, signature);
}
```

```js
// File: src/validate.js
import { MODELS, EFFORT_LEVELS, MAX_HISTORY, MAX_MESSAGE_CHARS } from "./config.js";

// クライアント入力を検証し、APIに渡せる形へ正規化する
export function validateChatBody(body) {
  if (!body || typeof body !== "object") {
    return { ok: false, error: "リクエスト形式が不正です。" };
  }

  const model = MODELS.find((m) => m.id === body.model);
  if (!model) {
    return { ok: false, error: "選択されたモデルは利用できません。" };
  }

  if (!Array.isArray(body.messages) || body.messages.length === 0) {
    return { ok: false, error: "メッセージが空です。" };
  }

  const sliced = body.messages.slice(-MAX_HISTORY);
  const messages = [];
  for (const msg of sliced) {
    if (!msg || (msg.role !== "user" && msg.role !== "assistant")) {
      return { ok: false, error: "メッセージ形式が不正です。" };
    }
    if (
      typeof msg.content !== "string" ||
      msg.content.trim() === "" ||
      msg.content.length > MAX_MESSAGE_CHARS
    ) {
      return { ok: false, error: "メッセージが空、または長すぎます（8000文字以内）。" };
    }
    // API の role 交互制約に合わせ、連続する同一 role は結合する
    const last = messages[messages.length - 1];
    if (last && last.role === msg.role) {
      last.content += "\n\n" + msg.content;
    } else {
      messages.push({ role: msg.role, content: msg.content });
    }
  }

  // 履歴スライスで先頭が assistant になった場合に備え、user 始まりに揃える
  while (messages.length > 0 && messages[0].role !== "user") messages.shift();
  if (messages.length === 0) {
    return { ok: false, error: "メッセージが空です。" };
  }

  // effort は対応モデルかつ有効な値のときだけ採用（それ以外は送らない）
  let effort = null;
  if (model.supportsEffort && typeof body.effort === "string" && EFFORT_LEVELS.includes(body.effort)) {
    effort = body.effort;
  }

  return { ok: true, value: { model, messages, effort } };
}
```

```js
// File: src/worker.js
import { MODELS, DEFAULT_MAX_TOKENS, EFFORT_BETA_HEADER } from "./config.js";
import {
  createSessionCookie,
  clearSessionCookie,
  verifyRequest,
  resolveAuthSecret,
  timingSafeEqual,
} from "./auth.js";
import { validateChatBody } from "./validate.js";

const API_URL = "https://api.anthropic.com/v1/messages";
const ANTHROPIC_VERSION = "2023-06-01";

function json(data, status = 200, extraHeaders = {}) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      "Content-Type": "application/json; charset=utf-8",
      "Cache-Control": "no-store",
      "X-Content-Type-Options": "nosniff",
      ...extraHeaders,
    },
  });
}

// 上流のエラー内容（APIキー等の内部情報を含みうる）は画面に返さず、状況別の定型文に変換する
function upstreamErrorMessage(status) {
  if (status === 401 || status === 403) {
    return "サーバー側の認証設定に問題があります。管理者に連絡してください。";
  }
  if (status === 400) {
    return "リクエストが拒否されました。入力を短くするか、モデルを変更してください。";
  }
  if (status === 404) {
    return "指定されたモデルが見つかりません。管理者に連絡してください。";
  }
  if (status === 429) {
    return "利用制限に達しました。しばらく待ってから再試行してください。";
  }
  if (status === 529 || status === 503) {
    return "Claude API が混雑しています。時間をおいて再試行してください。";
  }
  return "Claude API で一時的なエラーが発生しました。時間をおいて再試行してください。";
}

async function requireAuth(request, env) {
  return verifyRequest(request, resolveAuthSecret(env));
}

async function handleLogin(request, env) {
  let body = null;
  try {
    body = await request.json();
  } catch {
    return json({ error: "リクエスト形式が不正です。" }, 400);
  }

  const secret = resolveAuthSecret(env);
  if (!secret) {
    return json({ error: "サーバー設定エラーです。管理者に連絡してください。" }, 500);
  }

  const expected = env.ACCESS_CODE || "1359";
  const code = typeof body?.code === "string" ? body.code : "";
  if (!timingSafeEqual(code, expected)) {
    // 総当たり緩和のため失敗時は少し待つ
    await new Promise((resolve) => setTimeout(resolve, 400));
    return json({ error: "アクセスコードが違います。" }, 401);
  }

  return json({ ok: true }, 200, { "Set-Cookie": await createSessionCookie(secret) });
}

async function handleChat(request, env) {
  if (!env.CLAUDE_API_KEY) {
    return json({ error: "サーバー設定エラーです。管理者に連絡してください。" }, 500);
  }

  let body = null;
  try {
    body = await request.json();
  } catch {
    return json({ error: "リクエスト形式が不正です。" }, 400);
  }

  const check = validateChatBody(body);
  if (!check.ok) {
    return json({ error: check.error }, 400);
  }
  const { model, messages, effort } = check.value;

  const payload = {
    model: model.id,
    max_tokens: DEFAULT_MAX_TOKENS,
    stream: true,
    messages,
  };
  const headers = {
    "Content-Type": "application/json",
    "x-api-key": env.CLAUDE_API_KEY,
    "anthropic-version": ANTHROPIC_VERSION,
  };
  if (effort) {
    payload.output_config = { effort };
    headers["anthropic-beta"] = EFFORT_BETA_HEADER;
  }

  let upstream;
  try {
    upstream = await fetch(API_URL, {
      method: "POST",
      headers,
      body: JSON.stringify(payload),
    });
  } catch (err) {
    console.error("upstream fetch failed", err);
    return json({ error: "Claude API に接続できませんでした。時間をおいて再試行してください。" }, 502);
  }

  if (!upstream.ok || !upstream.body) {
    const status = upstream.status;
    // 詳細はサーバーログにのみ残し、クライアントには返さない
    console.error("anthropic api error", status, await upstream.text().catch(() => ""));
    return json({ error: upstreamErrorMessage(status) }, status === 429 ? 429 : 502);
  }

  return new Response(upstream.body, {
    status: 200,
    headers: {
      "Content-Type": "text/event-stream; charset=utf-8",
      "Cache-Control": "no-store",
      "X-Content-Type-Options": "nosniff",
    },
  });
}

export default {
  async fetch(request, env) {
    const { pathname } = new URL(request.url);

    if (pathname === "/api/login" && request.method === "POST") {
      return handleLogin(request, env);
    }
    if (pathname === "/api/logout" && request.method === "POST") {
      return json({ ok: true }, 200, { "Set-Cookie": clearSessionCookie() });
    }
    if (pathname === "/api/session" && request.method === "GET") {
      return (await requireAuth(request, env))
        ? json({ ok: true })
        : json({ error: "unauthorized" }, 401);
    }
    if (pathname === "/api/models" && request.method === "GET") {
      if (!(await requireAuth(request, env))) {
        return json({ error: "ログインが必要です。" }, 401);
      }
      return json({
        models: MODELS.map(({ id, label, supportsEffort }) => ({ id, label, supportsEffort })),
      });
    }
    if (pathname === "/api/chat" && request.method === "POST") {
      if (!(await requireAuth(request, env))) {
        return json({ error: "ログインが必要です。" }, 401);
      }
      return handleChat(request, env);
    }
    if (pathname.startsWith("/api/")) {
      return json({ error: "Not Found" }, 404);
    }
    // 静的アセットに一致しないパス
    return new Response("Not Found", { status: 404 });
  },
};
```

```js
// File: public/app.js
(() => {
  "use strict";

  const loginView = document.getElementById("login-view");
  const chatView = document.getElementById("chat-view");
  const loginForm = document.getElementById("login-form");
  const loginCode = document.getElementById("login-code");
  const loginError = document.getElementById("login-error");
  const loginButton = document.getElementById("login-button");
  const modelSelect = document.getElementById("model-select");
  const effortField = document.getElementById("effort-field");
  const effortSelect = document.getElementById("effort-select");
  const logoutButton = document.getElementById("logout-button");
  const messagesEl = document.getElementById("messages");
  const emptyStateEl = document.getElementById("empty-state");
  const composer = document.getElementById("composer");
  const inputEl = document.getElementById("input");
  const sendButton = document.getElementById("send-button");
  const copyInputButton = document.getElementById("copy-input-button");

  const state = { messages: [], models: [], streaming: false };

  /* ---------- Markdown ---------- */
  function renderMarkdown(text) {
    if (window.marked && window.DOMPurify) {
      return window.DOMPurify.sanitize(window.marked.parse(text, { breaks: true, gfm: true }));
    }
    // CDN 失敗時のフォールバック: プレーンテキストとして表示
    const tmp = document.createElement("div");
    tmp.textContent = text;
    return tmp.innerHTML.replace(/\n/g, "<br>");
  }

  /* ---------- ユーティリティ ---------- */
  function scrollToBottom() {
    const main = document.querySelector(".chat-main");
    main.scrollTop = main.scrollHeight;
  }

  async function copyText(text, button) {
    if (!text) return;
    let done = false;
    try {
      await navigator.clipboard.writeText(text);
      done = true;
    } catch {
      const ta = document.createElement("textarea");
      ta.value = text;
      ta.style.position = "fixed";
      ta.style.opacity = "0";
      document.body.appendChild(ta);
      ta.select();
      try {
        done = document.execCommand("copy");
      } catch {
        done = false;
      }
      ta.remove();
    }
    if (button) {
      if (!button.dataset.label) button.dataset.label = button.textContent;
      button.textContent = done ? "コピーしました" : "コピーに失敗しました";
      setTimeout(() => {
        button.textContent = button.dataset.label;
      }, 1500);
    }
  }

  function showToast(message) {
    const toast = document.createElement("div");
    toast.className = "toast";
    toast.textContent = message;
    document.body.appendChild(toast);
    setTimeout(() => toast.remove(), 4000);
  }

  /* ---------- メッセージ描画 ---------- */
  function buildMessage(role, initialText) {
    const wrap = document.createElement("article");
    wrap.className = `message message-${role}`;

    const meta = document.createElement("div");
    meta.className = "message-meta";
    const label = document.createElement("span");
    label.textContent = role === "user" ? "あなた" : "Claude";
    meta.appendChild(label);

    const body = document.createElement("div");
    body.className = "message-body";
    wrap.append(meta, body);

    let raw = initialText || "";
    const render = () => {
      body.innerHTML = renderMarkdown(raw);
    };

    if (role === "assistant") {
      const copyBtn = document.createElement("button");
      copyBtn.type = "button";
      copyBtn.className = "copy-button";
      copyBtn.textContent = "コピー";
      copyBtn.addEventListener("click", () => copyText(raw, copyBtn));
      meta.appendChild(copyBtn);
    }

    render();
    emptyStateEl.hidden = true;
    messagesEl.appendChild(wrap);
    scrollToBottom();

    return {
      append(delta) {
        raw += delta;
      },
      render,
      get raw() {
        return raw;
      },
      showError(message) {
        const err = document.createElement("p");
        err.className = "inline-error";
        err.textContent = message;
        wrap.appendChild(err);
        scrollToBottom();
      },
      remove() {
        wrap.remove();
      },
    };
  }

  /* ---------- SSE ストリーミング ---------- */
  async function streamResponse(body, onDelta) {
    const reader = body.getReader();
    const decoder = new TextDecoder();
    let buffer = "";
    let streamError = "";

    const handleEvent = (rawEvent) => {
      const dataLines = [];
      for (const line of rawEvent.split(/\r?\n/)) {
        if (line.startsWith("data:")) dataLines.push(line.slice(5).trimStart());
      }
      if (!dataLines.length) return;
      let data;
      try {
        data = JSON.parse(dataLines.join("\n"));
      } catch {
        return;
      }
      if (data.type === "content_block_delta" && data.delta && data.delta.type === "text_delta") {
        onDelta(data.delta.text);
      } else if (data.type === "error") {
        streamError = (data.error && data.error.type) || "unknown";
      }
    };

    for (;;) {
      const { done, value } = await reader.read();
      if (done) break;
      buffer += decoder.decode(value, { stream: true });
      const events = buffer.split(/\r?\n\r?\n/);
      buffer = events.pop();
      for (const ev of events) handleEvent(ev);
    }
    if (buffer.trim()) handleEvent(buffer);
    if (streamError) {
      throw new Error("応答の途中でエラーが発生しました。もう一度お試しください。");
    }
  }

  /* ---------- 送信 ---------- */
  function setStreaming(flag) {
    state.streaming = flag;
    sendButton.disabled = flag;
    sendButton.textContent = flag ? "送信中…" : "送信";
  }

  function autoResize() {
    inputEl.style.height = "auto";
    inputEl.style.height = Math.min(inputEl.scrollHeight, 200) + "px";
  }

  async function handleSend() {
    const text = inputEl.value.trim();
    if (!text || state.streaming) return;

    inputEl.value = "";
    autoResize();
    const userMsg = buildMessage("user", text);
    state.messages.push({ role: "user", content: text });
    const assistantMsg = buildMessage("assistant", "");
    setStreaming(true);

    let renderScheduled = false;
    const scheduleRender = () => {
      if (renderScheduled) return;
      renderScheduled = true;
      requestAnimationFrame(() => {
        renderScheduled = false;
        assistantMsg.render();
        scrollToBottom();
      });
    };

    let received = false;
    try {
      const res = await fetch("/api/chat", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          messages: state.messages,
          model: modelSelect.value,
          effort: effortSelect.disabled ? undefined : effortSelect.value,
        }),
      });

      if (res.status === 401) {
        rollback();
        showLoginView("セッションが切れました。再度ログインしてください。");
        return;
      }
      if (!res.ok || !res.body) {
        const data = await res.json().catch(() => null);
        throw new Error((data && data.error) || "エラーが発生しました。時間をおいて再試行してください。");
      }

      await streamResponse(res.body, (delta) => {
        received = true;
        assistantMsg.append(delta);
        scheduleRender();
      });

      assistantMsg.render();
      scrollToBottom();
      state.messages.push({ role: "assistant", content: assistantMsg.raw });
    } catch (err) {
      if (!received) {
        // 送信自体の失敗: ユーザーメッセージを取り消して入力を復元
        rollback();
        showToast(err.message || "エラーが発生しました。");
      } else {
        // 途中まで届いている場合は部分応答を残してエラーを併記
        assistantMsg.render();
        state.messages.push({ role: "assistant", content: assistantMsg.raw });
        assistantMsg.showError(err.message || "ストリーミング中にエラーが発生しました。");
      }
    } finally {
      setStreaming(false);
      inputEl.focus();
    }

    function rollback() {
      state.messages.pop();
      userMsg.remove();
      assistantMsg.remove();
      inputEl.value = text;
      autoResize();
    }
  }

  /* ---------- 画面切り替え / 初期化 ---------- */
  function showLoginView(message) {
    chatView.hidden = true;
    loginView.hidden = false;
    if (message) loginError.textContent = message;
    loginCode.focus();
  }

  function updateEffortState() {
    const model = state.models.find((m) => m.id === modelSelect.value);
    const supported = Boolean(model && model.supportsEffort);
    effortSelect.disabled = !supported;
    effortField.classList.toggle("is-disabled", !supported);
    effortField.title = supported ? "" : "このモデルでは Effort は利用できません";
  }

  async function loadModels() {
    const res = await fetch("/api/models");
    if (!res.ok) return;
    const { models } = await res.json();
    state.models = models;
    modelSelect.innerHTML = "";
    for (const m of models) {
      const opt = document.createElement("option");
      opt.value = m.id;
      opt.textContent = m.label;
      modelSelect.appendChild(opt);
    }
    updateEffortState();
  }

  async function enterChat() {
    loginView.hidden = true;
    chatView.hidden = false;
    await loadModels();
    inputEl.focus();
  }

  /* ---------- イベント ---------- */
  loginForm.addEventListener("submit", async (e) => {
    e.preventDefault();
    loginError.textContent = "";
    loginButton.disabled = true;
    try {
      const res = await fetch("/api/login", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ code: loginCode.value }),
      });
      if (res.ok) {
        loginCode.value = "";
        await enterChat();
      } else {
        const data = await res.json().catch(() => null);
        loginError.textContent = (data && data.error) || "ログインに失敗しました。";
      }
    } catch {
      loginError.textContent = "通信エラーが発生しました。";
    } finally {
      loginButton.disabled = false;
      loginCode.focus();
    }
  });

  logoutButton.addEventListener("click", async () => {
    await fetch("/api/logout", { method: "POST" }).catch(() => {});
    location.reload();
  });

  modelSelect.addEventListener("change", updateEffortState);

  composer.addEventListener("submit", (e) => {
    e.preventDefault();
    handleSend();
  });

  inputEl.addEventListener("keydown", (e) => {
    // 日本語IME変換中の Enter では送信しない
    if (e.key === "Enter" && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      handleSend();
    }
  });

  inputEl.addEventListener("input", autoResize);

  copyInputButton.addEventListener("click", () => copyText(inputEl.value, copyInputButton));

  /* ---------- 起動 ---------- */
  (async function init() {
    try {
      const res = await fetch("/api/session");
      if (res.ok) {
        await enterChat();
        return;
      }
    } catch {
      // ネットワーク不通時もログイン画面を表示
    }
    showLoginView();
  })();
})();
```

````markdown
<!-- File: README.md -->
# Claude Chat

Cloudflare Workers 上で動く、Claude API 利用のシンプルなチャット Web アプリです。
API キーはサーバー側のシークレットとして管理され、ブラウザには一切露出しません。

## 機能

- 簡易ログイン（アクセスコード方式。デフォルト: `1359`）
- モデル選択: Claude Haiku 4.5 / Claude Sonnet 4.5 / Claude Opus 4.5
- Effort（推論量）選択: Opus 4.5 のみ Low / Medium / High
- ストリーミング表示（SSE）、チャット履歴表示
- 入出力ともに Markdown レンダリング（コードブロック・箇条書き・表など）
- 入力欄・各回答にワンクリックのコピーボタン
- PC / スマホ対応の落ち着いた UI

## 技術構成

- Cloudflare Workers（API プロキシ + 認証）+ Workers Static Assets（フロントエンド配信）
- ビルド不要。依存は開発用の `wrangler` のみ
- Markdown: `marked` + `DOMPurify`（CDN 読み込み。失敗時はプレーンテキストにフォールバック）
- テスト: Node.js 標準の `node:test`

## モデルと Effort について（一次情報に基づく判断）

- モデルIDは Anthropic 公式ドキュメント「Models overview」に基づく
  - Claude Haiku 4.5: `claude-haiku-4-5-20251001`
  - Claude Sonnet 4.5: `claude-sonnet-4-5-20250929`
  - Claude Opus 4.5: `claude-opus-4-5-20251101`
- 要件で挙がっていた「Sonnet 5 / Opus 5」は実装時点で公式に提供されていないため、
  利用可能な最新モデル（Sonnet 4.5 / Opus 4.5）に置き換えている。
  新モデルが公開された場合は `src/config.js` の `MODELS` 配列だけを更新すればよい。
- Effort は公式ドキュメント「Effort」に基づき Opus 4.5 のみ有効化している
  - `output_config: { effort: "low" | "medium" | "high" }` を送信
  - beta ヘッダー `anthropic-beta: effort-2025-11-24` を付与
  - 非対応モデルでは UI 上で選択肢を無効化し、サーバー側でも effort を送信しない

## 前提

- Node.js 20 以上
- Cloudflare アカウント
- Anthropic API キー（Console で発行）

## ローカルでの起動方法

1. 依存関係のインストール

       npm install

2. 環境変数ファイルの作成

       cp .dev.vars.example .dev.vars

   `.dev.vars` を編集し、以下を設定する

       CLAUDE_API_KEY=sk-ant-...        # 必須
       ACCESS_CODE=1359                 # 任意（未設定時は 1359）
       AUTH_SECRET=ランダムな長い文字列  # 推奨

3. 開発サーバー起動

       npm run dev

4. 表示された URL（通常は `http://localhost:8787`）にアクセスし、
   アクセスコード `1359` でログインする

   なお、ログイン画面には意地悪な注意書きとして
   「アルファベット大文字小文字記号を含む」と表示されるが、
   実際の判定条件は `ACCESS_CODE`（デフォルト `1359`）との一致のみである。

## Cloudflare へのデプロイ手順

1. Cloudflare にログイン

       npx wrangler login

2. シークレットの登録（対話プロンプトで値を入力）

       npx wrangler secret put CLAUDE_API_KEY
       npx wrangler secret put AUTH_SECRET
       npx wrangler secret put ACCESS_CODE

   - `CLAUDE_API_KEY` は必須
   - `AUTH_SECRET` は推奨（未設定時は `CLAUDE_API_KEY` からセッション署名鍵を派生）
   - `ACCESS_CODE` は任意（未設定時は `1359`）

3. デプロイ

       npm run deploy

4. 発行された `https://<name>.<account>.workers.dev` にアクセスして動作確認

   カスタムドメインを使う場合は `wrangler.toml` に `routes` を追加するか、
   ダッシュボードから設定する。

## 環境変数一覧

| 名前 | 種別 | 必須 | 説明 |
| --- | --- | --- | --- |
| `CLAUDE_API_KEY` | secret | 必須 | Anthropic API キー。サーバー側のみで使用 |
| `ACCESS_CODE` | secret | 任意 | ログイン用アクセスコード。未設定時は `1359` |
| `AUTH_SECRET` | secret | 推奨 | セッション Cookie の署名鍵。未設定時は `CLAUDE_API_KEY` で代用 |

## テスト

    npm test

`node:test` で認証（署名・改ざん・期限切れ）とリクエスト検証
（モデル許可リスト、effort の適用条件、role 正規化など）をカバーしている。

## セキュリティ上の注意

- 本アプリの認証はあくまで簡易的なアクセス制限。
  本格的な保護が必要な場合は Cloudflare Access などの利用を推奨する。
- API キーは Workers のシークレットにのみ存在し、
  クライアントのコード・レスポンス・エラーメッセージには含まれない。
- 上流 API のエラー詳細はサーバーログにのみ出力し、
  画面には状況別の定型メッセージのみ表示する。
- セッション Cookie は `HttpOnly; Secure; SameSite=Strict`、有効期限 7 日。
- CSP 等のセキュリティヘッダーは `public/_headers` で付与している。

## トラブルシューティング

- 「サーバー側の認証設定に問題があります」と出る
  → `CLAUDE_API_KEY` が未設定か無効。`wrangler secret put CLAUDE_API_KEY` で再設定。
- 「利用制限に達しました」と出る
  → Anthropic 側のレート制限。時間をおくか利用プランを確認。
- Markdown がレンダリングされない
  → CDN（jsdelivr）への接続を確認。接続できない場合はプレーンテキストで表示される仕様。
  完全なオフライン動作が必要な場合は marked / DOMPurify を `public/vendor/` に同梱し、
  `index.html` の script タグと CSP を書き換える。
````
